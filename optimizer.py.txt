"""
Energy Detective AI
File 3: optimizer.py

Search-based scheduling optimizer for flexible household appliances.
Uses coordinate descent with feasible-move checks.
"""

from energy_tools import (
    build_load_profile,
    calculate_schedule_metrics,
    check_constraints,
)


def _objective(
    appliances,
    starts,
    prices,
    solar,
    base_load_kw,
    carbon_intensity,
    demand_charge,
    carbon_weight,
):
    """Return the modeled objective value for a schedule."""

    metrics = calculate_schedule_metrics(
        appliances=appliances,
        starts=starts,
        prices=prices,
        solar=solar,
        base_load_kw=base_load_kw,
        carbon_intensity=carbon_intensity,
        demand_charge=demand_charge,
        carbon_weight=carbon_weight,
    )

    return metrics["objective"], metrics


def optimize_schedule(
    appliances,
    initial_starts,
    prices,
    solar,
    base_load_kw,
    carbon_intensity,
    demand_charge,
    carbon_weight,
    grid_limit,
):
    """
    Improve the appliance schedule through local search.

    Each candidate moves one appliance to another permitted start time.
    A candidate is accepted only if it satisfies the configured
    constraints and improves the objective.
    """

    starts = dict(initial_starts)
    log = []

    def feasible(candidate_starts):
        load = build_load_profile(
            appliances=appliances,
            starts=candidate_starts,
            base_load_kw=base_load_kw,
        )

        violations = check_constraints(
            appliances=appliances,
            starts=candidate_starts,
            load_profile=load,
            grid_limit=grid_limit,
        )

        return len(violations) == 0

    # If the original schedule violates constraints, search for
    # a feasible starting plan before attempting cost improvements.
    if not feasible(starts):
        found_feasible = False

        # Start from the most flexible jobs first.
        ordered = sorted(
            appliances,
            key=lambda item: (
                item["latest_start"] - item["earliest_start"]
            ),
            reverse=True,
        )

        def find_feasible(index):
            nonlocal starts, found_feasible

            if index >= len(ordered):
                if feasible(starts):
                    found_feasible = True
                    return True
                return False

            appliance = ordered[index]
            appliance_id = appliance["id"]

            original = starts[appliance_id]

            candidates = list(range(
                appliance["earliest_start"],
                appliance["latest_start"] + 1,
            ))

            candidates.sort(
                key=lambda hour: abs(hour - original)
            )

            for hour in candidates:
                starts[appliance_id] = hour

                if find_feasible(index + 1):
                    return True

            starts[appliance_id] = original
            return False

        find_feasible(0)

        if not found_feasible:
            # Return the original plan if no feasible plan is found.
            # The agent will report remaining violations.
            starts = dict(initial_starts)

            log.append({
                "appliance": "System",
                "from": "Original plan",
                "to": "No feasible plan found",
                "reason": (
                    "The configured operating windows and grid limit "
                    "did not yield a feasible schedule during the search."
                ),
            })

            return starts, log

        log.append({
            "appliance": "System",
            "from": "Infeasible plan",
            "to": "Feasible starting plan",
            "reason": (
                "The optimizer found a schedule that satisfies "
                "the configured constraints."
            ),
        })

    current_value, current_metrics = _objective(
        appliances=appliances,
        starts=starts,
        prices=prices,
        solar=solar,
        base_load_kw=base_load_kw,
        carbon_intensity=carbon_intensity,
        demand_charge=demand_charge,
        carbon_weight=carbon_weight,
    )

    # Repeatedly search for improving single-appliance moves.
    max_passes = 30
    improvement_tolerance = 0.0001

    for pass_number in range(1, max_passes + 1):
        improved_this_pass = False

        # Try flexible, high-energy appliances first.
        ordered = sorted(
            appliances,
            key=lambda item: (
                item["power"] * item["duration"]
            ),
            reverse=True,
        )

        for appliance in ordered:
            appliance_id = appliance["id"]
            old_hour = starts[appliance_id]

            best_hour = old_hour
            best_value = current_value
            best_metrics = current_metrics

            candidates = list(range(
                appliance["earliest_start"],
                appliance["latest_start"] + 1,
            ))

            # Evaluate cheaper-looking periods first.
            candidates.sort(
                key=lambda hour: (
                    sum(
                        prices[(hour + offset) % 24]
                        for offset in range(appliance["duration"])
                    ),
                    hour,
                )
            )

            for candidate_hour in candidates:
                if candidate_hour == old_hour:
                    continue

                candidate_starts = dict(starts)
                candidate_starts[appliance_id] = candidate_hour

                if not feasible(candidate_starts):
                    continue

                candidate_value, candidate_metrics = _objective(
                    appliances=appliances,
                    starts=candidate_starts,
                    prices=prices,
                    solar=solar,
                    base_load_kw=base_load_kw,
                    carbon_intensity=carbon_intensity,
                    demand_charge=demand_charge,
                    carbon_weight=carbon_weight,
                )

                if (
                    candidate_value
                    < best_value - improvement_tolerance
                ):
                    best_hour = candidate_hour
                    best_value = candidate_value
                    best_metrics = candidate_metrics

            if best_hour != old_hour:
                old_cost = current_metrics["total_cost"]
                new_cost = best_metrics["total_cost"]

                starts[appliance_id] = best_hour
                current_value = best_value
                current_metrics = best_metrics

                log.append({
                    "appliance": appliance["name"],
                    "from": f"{old_hour:02d}:00",
                    "to": f"{best_hour:02d}:00",
                    "reason": (
                        f"Pass {pass_number}: the move improved "
                        f"the objective from "
                        f"{current_value:.4f} or less. "
                        f"Modeled bill after move: "
                        f"${new_cost:.2f}."
                    ),
                    "previous_bill": round(old_cost, 4),
                    "new_bill": round(new_cost, 4),
                })

                improved_this_pass = True

        if not improved_this_pass:
            break

    return starts, log
